cmake_minimum_required(VERSION 3.16)
project(fulldex_national_dex_checks LANGUAGES C CXX)
enable_testing()
find_package(Threads REQUIRED)
include(../../cmake/PokemonHub.cmake)
add_executable(dex_model ../test_national_dex.cpp)
target_compile_features(dex_model PRIVATE cxx_std_17)
target_include_directories(dex_model PRIVATE ../../frontend)
target_link_libraries(dex_model PRIVATE Threads::Threads)
add_test(NAME dex_model COMMAND dex_model)
add_executable(dex_readers ../test_national_dex_readers.cpp)
target_compile_features(dex_readers PRIVATE cxx_std_20)
target_include_directories(dex_readers PRIVATE ../../frontend ${PKSMCORE_INCLUDE_DIRS} ../../third_party/pksmcore-config)
target_link_libraries(dex_readers PRIVATE pokemonhub)
add_test(NAME dex_readers COMMAND dex_readers)
# All assertions stay active in Release builds too.
foreach(target dex_model dex_readers)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endforeach()
add_executable(dex_art ../test_national_dex_art.cpp)
target_compile_features(dex_art PRIVATE cxx_std_17)
target_include_directories(dex_art PRIVATE ../../frontend)
if(MSVC)
    target_compile_options(dex_art PRIVATE /UNDEBUG)
else()
    target_compile_options(dex_art PRIVATE -UNDEBUG)
endif()
add_test(NAME dex_art COMMAND dex_art)
foreach(mode enabled disabled)
    add_executable(dex_ui_${mode} ../test_national_dex_ui.cpp ../../frontend/launcher_cjk.cpp)
    target_compile_features(dex_ui_${mode} PRIVATE cxx_std_17)
    target_include_directories(dex_ui_${mode} PRIVATE ../../frontend)
    target_link_libraries(dex_ui_${mode} PRIVATE Threads::Threads)
    if(mode STREQUAL "enabled")
        target_compile_definitions(dex_ui_${mode} PRIVATE SSR_POKEMON_HUB)
        target_link_libraries(dex_ui_${mode} PRIVATE pokemonhub)
    endif()
    if(MSVC)
        target_compile_options(dex_ui_${mode} PRIVATE /UNDEBUG)
    else()
        target_compile_options(dex_ui_${mode} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME dex_ui_${mode} COMMAND dex_ui_${mode})
endforeach()
